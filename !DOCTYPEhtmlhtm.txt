    const starCanvas = document.getElementById('starCanvas');
    const starCtx = starCanvas.getContext('2d');
    starCanvas.width = window.innerWidth;
    starCanvas.height = window.innerHeight;

    const stars = [];
    for (let i = 0; i < 100; i++) {
      stars.push({
        x: Math.random() * starCanvas.width,
        y: Math.random() * starCanvas.height,
        size: Math.random() * 2 + 1,
        opacity: Math.random(),
        speed: Math.random() * 0.5
      });
    }

    function drawStars() {
      starCtx.clearRect(0, 0, starCanvas.width, starCanvas.height);
      stars.forEach(star => {
        starCtx.fillStyle = `rgba(255, 255, 255, ${star.opacity})`;
        starCtx.fillRect(star.x, star.y, star.size, star.size);
        star.opacity += (Math.random() - 0.5) * 0.05;
        star.opacity = Math.max(0.3, Math.min(1, star.opacity));
        star.y += star.speed;
        if (star.y > starCanvas.height) {
          star.y = 0;
          star.x = Math.random() * starCanvas.width;
        }
      });
      requestAnimationFrame(drawStars);
    }
    drawStars();

    setInterval(() => {
      const meteor = {
        x: Math.random() * starCanvas.width,
        y: 0,
        speed: 8
      };
      const meteorInterval = setInterval(() => {
        starCtx.fillStyle = '#FFD700';
        starCtx.fillRect(meteor.x, meteor.y, 4, 4);

    let currentDetail = null;

    function openDetail(level) {
      currentDetail = level;
      document.getElementById('mainLobby').classList.add('hidden');
      document.getElementById(`detail${level}`).classList.add('active');
      showMessage(`detail${level}`);

      const numbers = document.querySelectorAll(`#detail${level} .number-value[data-target]`);
      numbers.forEach(num => {
        const target = parseInt(num.dataset.target);
        let current = 0;
        const increment = target / 50;
        const timer = setInterval(() => {
          current += increment;
          if (current >= target) {
            num.textContent = target;
            clearInterval(timer);
          } else {
            num.textContent = Math.floor(current);
          }
        }, 30);
      });
    }

    function closeDetail() {
      if (currentDetail) {
        document.getElementById(`detail${currentDetail}`).classList.remove('active');
        document.getElementById('mainLobby').classList.remove('hidden');
        currentDetail = null;
        showMessage('back');
      }
    }
